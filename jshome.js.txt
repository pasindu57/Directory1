/* ============================================================
   BODIMA – Homepage logic
   ============================================================ */
(function () {
  const props = getPublicProperties();

  /* ----- Hero search → redirects to Browse Properties ----- */
  const heroForm = qs('#heroSearch');
  if (heroForm) {
    qs('#heroType').innerHTML =
      '<option value="">Any type</option>' +
      CATEGORIES.map(c => `<option value="${c}">${c}</option>`).join('');

    heroForm.addEventListener('submit', e => {
      e.preventDefault();
      const params = new URLSearchParams();
      const loc = qs('#heroLocation').value.trim();
      const type = qs('#heroType').value;
      const max = qs('#heroBudget').value;
      if (loc) params.set('q', loc);
      if (type) params.set('type', type);
      if (max) params.set('max', max);
      location.href = 'properties.html' + (params.toString() ? '?' + params.toString() : '');
    });
  }

  /* ----- Hero stats (animated) ----- */
  const stats = qs('#heroStats');
  if (stats) {
    const districts = new Set(props.map(p => p.district)).size;
    const cities = new Set(props.map(p => p.city)).size;
    const items = [
      { icon: 'fa-house-chimney', num: props.length, label: 'Active Listings' },
      { icon: 'fa-map-location-dot', num: districts, label: 'Districts Covered' },
      { icon: 'fa-city', num: cities, label: 'Cities & Towns' },
      { icon: 'fa-layer-group', num: CATEGORIES.length, label: 'Categories' }
    ];
    stats.innerHTML = items.map(i =>
      `<div class="stat-chip"><i class="fa-solid ${i.icon}"></i>
        <b class="stat-num" data-count="${i.num}">0</b> ${i.label}</div>`).join('');
    initCounters();
  }

  /* ----- Featured listings ----- */
  const featuredGrid = qs('#featuredGrid');
  if (featuredGrid) {
    featuredGrid.innerHTML = [1, 2, 3, 4, 5, 6].map(() =>
      `<div class="skel-card"><div class="skel skel-img"></div>
        <div class="skel skel-line w80"></div><div class="skel skel-line w60"></div>
        <div class="skel skel-line w40"></div><div class="skel skel-pad"></div></div>`).join('');

    setTimeout(() => {
      const featured = props.filter(p => p.featured).slice(0, 6);
      featuredGrid.innerHTML = featured.map(renderPropCard).join('');
      initReveal(featuredGrid);
    }, 550);
  }

  /* ----- Categories with live counts ----- */
  const catGrid = qs('#categoryGrid');
  if (catGrid) {
    catGrid.innerHTML = CATEGORIES.map(cat => {
      const count = props.filter(p => p.type === cat).length;
      return `
      <a class="cat-card reveal" href="properties.html?type=${encodeURIComponent(cat)}">
        <span class="cat-icon"><i class="${CATEGORY_ICONS[cat]}"></i></span>
        <div class="cat-name">${cat}</div>
        <div class="cat-count">${count} listing${count !== 1 ? 's' : ''}</div>
      </a>`;
    }).join('');
  }

  /* ----- Popular locations ----- */
  const locGrid = qs('#locationGrid');
  if (locGrid) {
    locGrid.innerHTML = LOCATIONS.map(loc => {
      const count = props.filter(p => p.city === loc.name || p.district === loc.name).length;
      return `
      <a class="loc-card reveal" href="properties.html?q=${encodeURIComponent(loc.name)}">
        <img src="${IMG(loc.img, 600)}" alt="Accommodation in ${loc.name}" loading="lazy" onerror="imgFail(this)">
        <div class="loc-info">
          <div class="loc-name"><i class="fa-solid fa-location-dot"></i> ${loc.name}</div>
          <div class="loc-count">${count} place${count !== 1 ? 's' : ''} available</div>
        </div>
      </a>`;
    }).join('');
  }

  initReveal();
})();
